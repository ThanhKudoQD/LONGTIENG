NHIỆM VỤ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Bạn nhận 1 ARC của phim (đoạn cốt truyện lớn). Chia ARC này thành 2 cấp:

CẤP 1 — CHUNKS (như "chương" trong tiểu thuyết):
- Mỗi chunk ~{CHUNK_TARGET} dòng (target)
- Chunk là 1 mạch kịch có ý nghĩa, có tên gợi tả (vd: "Lần đầu gặp gỡ", "Đêm tân hôn")
- Mỗi arc có 3-{MAX_CHUNKS} chunks
- Dùng ARC_SUMMARY làm chỉ dẫn để chia chunks theo mạch truyện đúng

CẤP 2 — SCENES (cảnh con trong chunk):
- Chunk ≤ 100 dòng → KHÔNG chia scenes, scenes = [] (chunk = scene duy nhất)
- Chunk > 100 dòng → chia 3-10 scenes
- Mỗi scene là 1 mạch hội thoại liền, kết thúc khi:
  · Đổi nhóm nhân vật chính
  · Nhảy thời gian rõ (timestamp gap > 60s thường là cảnh mới)
  · Nhảy địa điểm xa
  · Đổi mục đích kịch

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
QUY TẮC CHIA
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

CHUNKS:
- Chunks LIỀN NHAU, không hở không lấn
- chunks[0].r[0] = ARC_START
- chunks[-1].r[1] = ARC_END
- chunks[i+1].r[0] = chunks[i].r[1] + 1

SCENES (trong mỗi chunk):
- Scenes LIỀN NHAU trong chunk
- scenes[0].r[0] = chunk.r[0]
- scenes[-1].r[1] = chunk.r[1]

TAG SCENE (optional):
- "HOOK": cảnh cliffhanger cuối arc / cuối chunk
- "PEAK": cảnh đỉnh cảm xúc trong chunk
- Bỏ trường tag nếu không có

EMOTION (chọn 1 cho mỗi scene):
neutral / happy / sad / angry / cold / tense / intimate /
fearful / sarcastic / shocked / determined / regretful / humorous / threatening

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
OUTPUT FORMAT (COMPACT)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Trả về JSON THUẦN, dùng array compact cho scenes để tiết kiệm token:

{
  "chunks": [
    {
      "r": [1, 250],
      "t": "Lần đầu gặp gỡ",
      "scenes": [
        [1, 35, ["顾沉舟", "苏念"], "tense", "office"],
        [36, 80, ["苏念", "李华"], "sad", "hospital"],
        [81, 250, ["顾沉舟", "苏念"], "intimate", "bedroom", "PEAK"]
      ]
    },
    {
      "r": [251, 500],
      "t": "Ký hợp đồng",
      "scenes": [
        [251, 500, ["顾沉舟", "苏念"], "tense", "office"]
      ]
    }
  ]
}

Mỗi scene = array với thứ tự cố định:
  [start_line, end_line, [characters_zh], emotion, location]
  hoặc
  [start_line, end_line, [characters_zh], emotion, location, "HOOK"|"PEAK"]
